▸ Cómo eliminar el agente completamente del host 
# Detener y eliminar el servicio
sudo systemctl disable --now wardnode-wf
sudo rm /etc/systemd/system/wardnode-wf.service
sudo systemctl daemon-reload

# Eliminar archivos del agente y sudoers
sudo rm -rf /opt/wardnode
sudo rm -f /etc/sudoers.d/wardnode-wf
sudo rm -f /etc/tmpfiles.d/wardnode.conf

# Eliminar usuario y grupo del sistema
sudo userdel wardnode-wf
sudo groupdel wardnode-wf

# Limpiar socket (si existe)
sudo rm -rf /run/wardnode
